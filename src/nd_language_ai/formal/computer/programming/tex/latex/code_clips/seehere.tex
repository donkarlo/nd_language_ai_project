\RequirePackage{xurl}

\makeatletter
\@ifpackageloaded{hyperref}{
    % ``colorlinks`` is a load-time option.  Reapplying it through
    % \hypersetup after hyperref is loaded produces a noisy warning.
    \hypersetup{
        linkcolor=blue,
        urlcolor=blue,
        citecolor=blue
    }
}{
    \RequirePackage[colorlinks=true]{hyperref}
    \hypersetup{
        linkcolor=blue,
        urlcolor=blue,
        citecolor=blue
    }
}
\makeatother

% Convert only absolute Linux filesystem paths to file:// URLs.
% Everything else keeps the old behavior.

\ExplSyntaxOn

\cs_new:Npn \nd_link_target:n #1
{
    \str_if_eq:eeTF
        { \tl_head:n {#1} }
        {/}
        { file://#1 }
        { #1 }
}

\NewDocumentCommand{\seehere}{m}
{
    \href{\nd_link_target:n{#1}}{see here}
}

\NewDocumentCommand{\clickurl}{m}
{
    \href{\nd_link_target:n{#1}}{url}
}

\NewDocumentCommand{\seeherefooter}{m}
{
    \href{\nd_link_target:n{#1}}{see here}
    \footnote{\tiny\url{\nd_link_target:n{#1}}}
}

\NewDocumentCommand{\clickurlfooter}{m}
{
    \href{\nd_link_target:n{#1}}{here}
    \footnote{\tiny\url{\nd_link_target:n{#1}}}
}

\ExplSyntaxOff
